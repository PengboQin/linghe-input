cmake_minimum_required(VERSION 3.24)
project(LingheAndroid LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
set(RIME "${ROOT}/build/rime/android-arm64")
add_library(linghe SHARED jni.cpp "${ROOT}/core/src/private_input.cpp" "${ROOT}/core/src/language_scoring.cpp" "${ROOT}/vendor/librime-octagram/src/gram_db.cc" "${ROOT}/vendor/librime-octagram/src/gram_encoding.cc" "${ROOT}/vendor/librime-octagram/src/octagram.cc")
target_include_directories(linghe PRIVATE "${ROOT}/core/include" "${RIME}/include" "${ROOT}/vendor/librime/src" "${ROOT}/vendor/librime/include" "${ROOT}/vendor/boost_1_85_0" "${ROOT}/build/android-librime/src")
target_link_libraries(linghe PRIVATE "-Wl,--whole-archive" "${RIME}/lib/librime.a" "-Wl,--no-whole-archive" "${RIME}/lib/libyaml-cpp.a" "${RIME}/lib/libleveldb.a" "${RIME}/lib/libopencc.a" "${RIME}/lib/libmarisa.a" log)
target_link_options(linghe PRIVATE -Wl,-z,max-page-size=16384)
