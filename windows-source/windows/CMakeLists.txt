cmake_minimum_required(VERSION 3.24)
project(PrivateInputWindows LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "TSF must be built against the Windows SDK on Windows")
endif()
set(CMAKE_CXX_STANDARD 17)
set(PRIVATE_RIME_PREFIX "" CACHE PATH "Pinned Rime installation for this Windows architecture")
if(NOT EXISTS "${PRIVATE_RIME_PREFIX}/include/rime_api.h")
  message(FATAL_ERROR "Provide PRIVATE_RIME_PREFIX with headers and same-architecture libraries")
endif()
add_library(PrivateInputTSF SHARED src/TextService.cpp activity/CaptureIPC.cpp activity/LearningIPC.cpp src/CandidateWindow.cpp src/Registration.cpp ../core/src/private_input.cpp PrivateInputTSF.def resources/PrivateInput.rc)
target_include_directories(PrivateInputTSF PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/resources ../core/include ${PRIVATE_RIME_PREFIX}/include)
target_compile_definitions(PrivateInputTSF PRIVATE UNICODE _UNICODE RIME_IMPORTS NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00 NTDDI_VERSION=0x0A000003)
target_link_directories(PrivateInputTSF PRIVATE ${PRIVATE_RIME_PREFIX}/lib)
if(EXISTS "${PRIVATE_RIME_PREFIX}/lib/librime.dll.a")
  set(PRIVATE_RIME_LINK "${PRIVATE_RIME_PREFIX}/lib/librime.dll.a")
else()
  find_library(PRIVATE_RIME_LINK NAMES rime REQUIRED PATHS "${PRIVATE_RIME_PREFIX}/lib" NO_DEFAULT_PATH)
endif()
find_library(PRIVATE_LEVELDB_LINK NAMES leveldb REQUIRED PATHS "${PRIVATE_RIME_PREFIX}/lib" NO_DEFAULT_PATH)
target_link_libraries(PrivateInputTSF PRIVATE "${PRIVATE_RIME_LINK}" "${PRIVATE_LEVELDB_LINK}" ole32 oleaut32 shell32 advapi32 user32 gdi32 uuid)
if(MSVC)
  target_compile_options(PrivateInputTSF PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(PrivateInputTSF PRIVATE -Wall -Wextra -Wno-unused-parameter -Wno-misleading-indentation)
endif()
set_target_properties(PrivateInputTSF PROPERTIES PREFIX "")

add_executable(LingheInputSetup WIN32 installer/Setup.cpp)
target_include_directories(LingheInputSetup PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src ../core/include)
target_compile_definitions(LingheInputSetup PRIVATE UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00)
target_link_libraries(LingheInputSetup PRIVATE shell32 ole32 advapi32 bcrypt user32 uuid)
if(MINGW)
  target_link_options(LingheInputSetup PRIVATE -municode)
endif()
if(MSVC)
  target_compile_options(LingheInputSetup PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(LingheInputSetup PRIVATE -Wall -Wextra -Wno-unused-parameter)
endif()

add_executable(LingheInputSettings WIN32 installer/Settings.cpp)
target_include_directories(LingheInputSettings PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src ../core/include)
target_compile_definitions(LingheInputSettings PRIVATE UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00)
target_link_libraries(LingheInputSettings PRIVATE shell32 ole32 user32 gdi32 advapi32 uuid)
if(MINGW)
  target_link_options(LingheInputSettings PRIVATE -municode)
endif()
if(MSVC)
  target_compile_options(LingheInputSettings PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(LingheInputSettings PRIVATE -Wall -Wextra -Wno-unused-parameter)

# The activity component is independent of the input thread and off by default.
set(PRIVATE_SODIUM_PREFIX "" CACHE PATH "Independent Windows libsodium prefix for activity")
if(PRIVATE_SODIUM_PREFIX)
 add_subdirectory(activity)
endif()

# Isolated, opt-in source-userdb export; a busy legacy DB is never copied.
add_executable(LingheLegacyExport WIN32 activity/LegacyExport.cpp activity/LegacyExport.rc ../core/src/private_input.cpp)
target_include_directories(LingheLegacyExport PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/activity" ../core/include ${PRIVATE_RIME_PREFIX}/include)
target_compile_definitions(LingheLegacyExport PRIVATE UNICODE _UNICODE RIME_IMPORTS NOMINMAX _WIN32_WINNT=0x0A00)
target_link_directories(LingheLegacyExport PRIVATE ${PRIVATE_RIME_PREFIX}/lib)
target_link_libraries(LingheLegacyExport PRIVATE "${PRIVATE_RIME_LINK}" "${PRIVATE_LEVELDB_LINK}" shell32 ole32 advapi32 user32 LingheActivityPlatform)
if(MINGW)
 target_link_options(LingheLegacyExport PRIVATE -municode -static)
endif()
